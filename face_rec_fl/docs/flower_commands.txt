# 6. RUN DEVICE SIMULATION (Cross-Device FL)
# Simulates 50 mobile devices with 20% participation per round (~10 active)
# 1 local epoch, batch size 32, 100 rounds, full dataset (no DEBUG_MAX_IDENTITIES)
# LR=0.05 with decay every 20 rounds
# Max 50 images per identity (simulates limited on-device storage)
export PYTORCH_CUDA_ALLOC_CONF=expandable_segments:True \
  CUDA_VISIBLE_DEVICES=7 \
  RAY_USAGE_STATS_ENABLED=0 \
  RAY_DISABLE_METRICS_EXPORT=1 \
  ENABLE_GLOBAL_EVAL=true; \
flwr run . --stream \
  --run-config "num-server-rounds=100 local-epochs=1 batch-size=32 learning-rate=0.05 fraction-train=0.2 max-imgs-per-identity=50 lr-decay-interval=20" \
  --federation-config "num-supernodes=50 backend.client-resources.num-cpus=4 backend.client-resources.num-gpus=0.1"
