# --------------------------------------------------------------------------
# C3: Few-Shot Device — ~6 hours
# 50 clients, 20% participation (10 active), 1 epoch, bs=32, 10 imgs/ID
# Simulates phones with very few photos per contact
# --------------------------------------------------------------------------
export PYTORCH_CUDA_ALLOC_CONF=expandable_segments:True \
  CUDA_VISIBLE_DEVICES=7 \
  RAY_USAGE_STATS_ENABLED=0 \
  RAY_DISABLE_METRICS_EXPORT=1 \
  ENABLE_GLOBAL_EVAL=true; \
flwr run . --stream \
  --run-config "num-server-rounds=100 local-epochs=1 batch-size=32 learning-rate=0.05 fraction-train=0.2 max-imgs-per-identity=10 lr-decay-interval=20" \
  --federation-config "num-supernodes=50 backend.client-resources.num-cpus=8 backend.client-resources.num-gpus=0.1"
